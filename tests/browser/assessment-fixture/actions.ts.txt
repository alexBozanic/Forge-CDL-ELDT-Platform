  _state: SubmissionState & { fixtureCount?: number },
  form: FormData,
) {
  const count = ++submissionCalls;
  await new Promise((resolve) => setTimeout(resolve, 4000));
  if (count === 1) throw new Error("PRIVATE SUBMISSION TRANSPORT DETAIL");
  const result = await submitAttempt(
    form,
    async () => ({
      data: { id, enrollment_id: id, status: "in_progress", questions },
      error: null,
    }),
    async () =>
      count < 3
        ? { data: null, error: { message: "PRIVATE FIXTURE DATABASE DETAIL" } }
        : { data: { status: "passed" }, error: null },
  );
  console.log(
    `Local submission fixture call ${count}: ${result.submitted ? "confirmed" : "error"}`,
  );
  return { ...result, fixtureCount: count };
